PySide6>=6.6.0

# --- Multimodal upgrade dependencies ---
# YOLO-World open-vocabulary object detection (pulls torch / torchvision automatically)

openai>=1.0.0

# Online TTS speech output (generated mp3 is played by the pygame mixer)
edge-tts

# TTS audio playback
pygame

# ASR speech recognition (Faster-Whisper, CPU + INT8 quantization, lazy-loaded on demand)
faster-whisper

# Live microphone recording (main.py uses sounddevice + wave)
sounddevice
